# 1. 创建 Secret
kubectl -n dev create secret generic redis-auth \
  --from-literal=password='%<@K0TKUo0K1oP&QZ;JvZLb['

# 2. 部署 ConfigMap
kubectl apply -f redis-config.yaml
kubectl apply -f sentinel-config.yaml

# 3. 部署 Service
kubectl apply -f redis-service.yaml
kubectl apply -f sentinel-service.yaml

# 4. 部署 Redis 和 Sentinel StatefulSet
kubectl apply -f redis-statefulset.yaml
kubectl apply -f sentinel-statefulset.yaml

# 5. 部署 Predixy
kubectl apply -f predixy-configmap.yaml
kubectl apply -f predixy-deployment.yaml

# 6. 部署 Gateway listener 和 TCPRoute
kubectl apply -f gateway-listener-patch.yaml
kubectl apply -f tcproute-redis.yaml


# 7. 验证
kubectl -n dev get pods -l app=redis
kubectl -n dev get pods -l app=sentinel
kubectl -n dev get pods -l app=predixy
kubectl -n dev get pvc | grep redis

# 8. 配置主从复制
# 主节点
kubectl -n dev exec redis-0 -- redis-cli INFO replication | grep -E "role|connected_slaves|slave[0-9]"
# 期望：role:master，connected_slaves:2，两个 slave state=online

# 从节点（等待同步完成）
kubectl -n dev exec redis-1 -- redis-cli INFO replication | grep role
# 期望：role:slave

kubectl -n dev exec sentinel-0 -- redis-cli -p 26379 SENTINEL master mymaster | grep -E "flags|num-slaves|num-other-sentinels"
# 期望：flags=master, num-slaves=2, num-other-sentinels=2

redis-cli -h <Gateway外部地址> -p 16379 -a '%<@K0TKUo0K1oP&QZ;JvZLb[' PING
# 期望：PONG

redis-cli -h <Gateway外部地址> -p 16379 -a '%<@K0TKUo0K1oP&QZ;JvZLb[' SET test "hello"
redis-cli -h <Gateway外部地址> -p 16379 -a '%<@K0TKUo0K1oP&QZ;JvZLb[' GET test


# 8. 故障转移测试
# 记录当前主节点（假设为 redis-0）
kubectl -n dev exec sentinel-0 -- redis-cli -p 26379 SENTINEL get-master-addr-by-name mymaster

# 删除主节点 Pod
kubectl -n dev delete pod redis-0

# 观察 Sentinel 日志
kubectl -n dev logs sentinel-0 -f | grep -E "sdown|odown|failover|promoted|switch-master"

# 等待约 10 秒后查询新主
kubectl -n dev exec sentinel-0 -- redis-cli -p 26379 SENTINEL get-master-addr-by-name mymaster
# 期望返回 redis-1 或 redis-2 的地址

# 通过 Predixy 写入验证（自动跟随新主）
redis-cli -h <Gateway外部地址> -p 16379 -a '%<@K0TKUo0K1oP&QZ;JvZLb[' SET after-failover "success"
# 期望：OK

# 10. 数据持久化验证
# 写入测试数据
kubectl -n dev exec redis-0 -- redis-cli SET persist-test "hello-ceph"

# 删除 Pod
kubectl -n dev delete pod redis-0

# 等待 Pod 重建后读取
kubectl -n dev exec redis-0 -- redis-cli GET persist-test
# 期望：hello-ceph（数据从 Ceph RBD 上的 AOF 恢复）
